do
    local function getSynchronizer()
        local ok, mod = pcall(require, ReplicatedStorage.Packages.Synchronizer)
        if ok then return mod end
        return nil
    end

    local function getTable(name)
        local sync = getSynchronizer()
        if not sync then return nil end
        local ok, data = pcall(sync.GetTableFromChannel, sync, name)
        if ok then return data end
        return nil
    end

    _G.XenSyncAll = function()
        return {}
    end

    _G.XenSyncGet = function(name)
        if type(name) ~= "string" then return nil end
        return getTable(name)
    end

    _G.sProp = function(ch, key)
        if type(ch) ~= "table" or key == nil then return nil end
        return ch[key]
    end

    local _AD, _MD, _TD
    local function _data()
        if _AD then return true end
        local ok = pcall(function()
            local d = ReplicatedStorage:WaitForChild("Datas")
            _AD = require(d:WaitForChild("Animals"))
            _MD = require(d:WaitForChild("Mutations"))
            _TD = require(d:WaitForChild("Traits"))
        end)
        return ok and _AD ~= nil
    end
    _G._xenGen = function(index, mutation, traits)
        if not _data() then return 0 end
        local info = _AD[index]
        if not info or not info.Generation then return 0 end
        local mult = 1
        if mutation and mutation ~= "None" and mutation ~= "" then
            local m = _MD[mutation]
            if m and m.Modifier then mult = mult + m.Modifier end
        end
        if type(traits) == "table" then
            for _, tr in ipairs(traits) do
                local t = _TD[tr]
                if t and t.MultiplierModifier then mult = mult + t.MultiplierModifier end
            end
        end
        return info.Generation * mult
    end
    _G._xenAnimShim = setmetatable({
        GetGeneration = function(_, index, mutation, traits)
            return _G._xenGen(index, mutation, traits)
        end
    }, {
        __index = function(_, k)
            local ok, real = pcall(function()
                return require(ReplicatedStorage:WaitForChild("Shared"):WaitForChild("Animals"))
            end)
            if ok and type(real) == "table" then return rawget(real, k) end
            return nil
        end
    })
end

do
    local RS = ReplicatedStorage
    local function resolveRemote(name)
        local pkgs = RS:FindFirstChild("Packages")
        local Net = pkgs and pkgs:FindFirstChild("Net")
        if not Net then return nil end
        local slots, byLeaf = {}, {}
        for i, c in ipairs(Net:GetChildren()) do
            local p, rest = string.match(c.Name, "^(%a+)/(.+)$")
            if p and #rest == 64 and string.match(rest, "^%x+$") then
                slots[i] = c
                byLeaf[rest] = byLeaf[rest] or {}
                table.insert(byLeaf[rest], { slot = i, class = c.ClassName })
            end
        end
        if name then
            for leaf, list in pairs(byLeaf) do
                if leaf == name then
                    for _, e in ipairs(list) do
                        if e.class == "RemoteEvent" then
                            return slots[e.slot]
                        end
                    end
                end
            end
            return nil
        end
        local anchor, dual = nil, 0
        for _, list in pairs(byLeaf) do
            if #list > 1 then
                dual = dual + 1
                for _, e in ipairs(list) do
                    if e.class == "RemoteEvent" then anchor = e.slot end
                end
            end
        end
        if dual ~= 1 or not anchor then return nil end
        local t = slots[anchor - 1]
        if t and t.ClassName == "RemoteEvent" and t.Parent == Net then return t end
        return nil
    end

    local remoteCache = {}
    local function getRemote(name)
        local key = name or "__UseItem"
        if remoteCache[key] and remoteCache[key].Parent then return remoteCache[key] end
        local r = resolveRemote(name)
        if r then remoteCache[key] = r end
        return r
    end

    _G.Net = {
        RemoteEvent = function(_, name) return getRemote(name) end,
        RemoteFunction = function(_, name) return nil end,
        UnreliableRemoteEvent = function(_, name) return nil end,
    }

    _G.sabcomFireGrapple2 = function(pos)
        local r = getRemote()
        if not r then return false end
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return false end
        pos = pos or (hrp.Position + workspace.CurrentCamera.CFrame.LookVector * 100)

        local a = math.clamp(dist / 120, 0, 2)
        return pcall(function()
            r:FireServer(a, pos)
        end)
    end
    _G.sabcomGrappleReady = function()
        return getRemote() ~= nil
    end
end

local function godMode(plr)
    plr.CharacterAdded:Connect(function(character)
        local humanoid = character:WaitForChild("Humanoid")
        humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
        humanoid:GetPropertyChangedSignal("Health"):Connect(function()
            if humanoid.Health < humanoid.MaxHealth then
                humanoid.Health = humanoid.MaxHealth
            end
        end)
        local deathConnection
        deathConnection = humanoid.Died:Connect(function()
            deathConnection:Disconnect()
            task.wait(0.05)
            if humanoid and humanoid.Parent then
                humanoid.Health = humanoid.MaxHealth
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
                deathConnection = humanoid.Died:Connect(function()
                    task.wait(0.05)
                    if humanoid and humanoid.Parent then
                        humanoid.Health = humanoid.MaxHealth
                        humanoid:ChangeState(Enum.HumanoidStateType.Running)
                    end
                end)
            end
        end)
    end)
end

local function createToggle(parent,label,order,def,green,key)
    local row=new("Frame",{Size=UDim2.new(1,0,0,17),BackgroundTransparency=1,LayoutOrder=order},parent)
    new("TextLabel",{Text=label,Font=FONT_BOLD,TextSize=11,TextColor3=Theme.Text,TextXAlignment=Enum.TextXAlignment.Left,BackgroundTransparency=1,Size=UDim2.new(0.7,0,1,0)},row)

    corner(track,9)
    local knob=new("Frame",{Size=UDim2.new(0,12,0,12),Position=UDim2.new(0,2.5,0.5,-6),BackgroundColor3=Color3.fromRGB(199,195,219)},track)
    corner(knob,7)
    local state={on=def}
    local function apply(anim)
        local onColor  = green and Theme.GreenDark or Theme.AccentB
        local kc       = state.on and Color3.new(1,1,1) or Color3.fromRGB(199,195,219)
        local tc       = state.on and onColor or Theme.PanelField2
        local kp       = state.on and UDim2.new(0,17.5,0.5,-6) or UDim2.new(0,2.5,0.5,-6)
        if anim then tw(track,{BackgroundColor3=tc},0.18); tw(knob,{Position=kp,BackgroundColor3=kc},0.18,Enum.EasingStyle.Back)
        else track.BackgroundColor3=tc; knob.Position=kp; knob.BackgroundColor3=kc end
        local g=track:FindFirstChildOfClass("UIGradient")
        if state.on then
            if not g then g=accentGrad(track,90) end
            g.Color=green and ColorSequence.new(Theme.Green,Theme.GreenDark) or ColorSequence.new(Theme.AccentA,Theme.AccentB)
        elseif g then g:Destroy() end
    end
    apply(false)
    track.MouseButton1Click:Connect(function() state.on=not state.on; apply(true); if key then Config[key]=state.on; saveConfig() end end)
    return {SetState=function(on) state.on=on; apply(true) end, GetState=function() return state.on end}
end

local allPetsCache = {}
local _PetPosCache = {}

local function getPetPosition(plot, slot)
    local key = plot.Name .. "|" .. tostring(slot)
    local hit = _PetPosCache[key]
    local now = os.clock()
    if hit and now < hit.exp then return hit.pos end

    local function compute()
        local podiums = plot:FindFirstChild("AnimalPodiums")
        if not podiums then return nil end
        local podium = podiums:FindFirstChild(tostring(slot))
        if not podium then return nil end
        for _, desc in ipairs(podium:GetDescendants()) do
            if desc:IsA("Model") and desc.Name ~= "Claim" and desc.Name ~= "Base" and desc.Name ~= "Decorations" then
                local hasMesh = false
                for _, c in ipairs(desc:GetDescendants()) do
                    if c:IsA("MeshPart") then hasMesh = true; break end
                end
                if hasMesh then
                    local ok, cf = pcall(function() return desc:GetBoundingBox() end)
                    if ok then return cf.Position end
                end
            end
        end
        local ok, cf = pcall(function() return podium:GetPivot() end)
        if ok then return cf.Position end
        return nil
    end

    local pos = compute()
    if pos then
        _PetPosCache[key] = { pos = pos, exp = now + 12 + math.random() * 8 }
    end
    return pos
end

local function resolveOwner(owner)
    if owner == nil then return nil end
    local plr
    pcall(function()
        if typeof(owner) == "Instance" then
            plr = owner:IsA("Player") and owner or Players:FindFirstChild(owner.Name)
        elseif type(owner) == "number" then
            plr = Players:GetPlayerByUserId(owner)
        elseif type(owner) == "string" then
            if tonumber(owner) then plr = Players:GetPlayerByUserId(tonumber(owner)) end
            if not plr then
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Name:lower() == owner:lower() or p.DisplayName:lower() == owner:lower() then
                        plr = p; break
                    end
                end
            end
        elseif type(owner) == "table" then
            if owner.UserId then plr = Players:GetPlayerByUserId(owner.UserId) end
            if not plr and owner.Name then plr = Players:FindFirstChild(tostring(owner.Name)) end
        end
    end)
    return plr
end

local function isOwnPlot(plotName)
    local ch = _G.XenSyncGet(plotName)
    if not ch then return false end
    local owner = _G.sProp(ch, "Owner")
    local plr = resolveOwner(owner)
    if plr then return plr.UserId == player.UserId end
    if type(owner) == "string" then
        local o = owner:lower()
        return o == player.Name:lower() or o == player.DisplayName:lower()
    end
    return false
end

local function ownerInGame(channel)
    if not channel then return false end
    local owner = _G.sProp(channel, "Owner")
    if not owner then return false end
    if resolveOwner(owner) then return true end
    if typeof(owner) == "Instance" or type(owner) == "number"
       or (type(owner) == "table" and (owner.UserId or owner.Name)) then
        return false
    end
    return true
end

local function scanAllPets()
    local pets = {}

    local plots = Workspace:FindFirstChild("Plots")

    for _, plot in ipairs(plots:GetChildren()) do
        if isOwnPlot(plot.Name) then continue end
        local ch = _G.XenSyncGet(plot.Name)
        if not ch or not ownerInGame(ch) then continue end

        local animalList = _G.sProp(ch, "AnimalList")
        if not animalList then continue end

        for slot, data in pairs(animalList) do
            if type(data) ~= "table" then continue end
            local name = data.Index
            if not name then continue end
            local info = AnimalsData and AnimalsData[name]
            if not info then continue end

            if type(data.Machine) == "table" and data.Machine.Active
               and ({Fuse=true, Duel=true, Trade=true, Crafting=true})[data.Machine.Type] then
                continue
            end

            local gen = 0
            pcall(function()
                gen = _G._xenAnimShim:GetGeneration(name, data.Mutation, data.Traits, nil)
            end)
            if gen <= 0 then continue end

            local pos = getPetPosition(plot, slot)
            if pos then
                table.insert(pets, {
                    name     = info.DisplayName or name,
                    index    = name,
                    mps      = gen,
                    mutation = data.Mutation or "None",
                    position = pos,
                    plot     = plot.Name,
                    slot     = tostring(slot),
                    uid      = plot.Name .. "_" .. tostring(slot),
                })
            end
        end
    end

    table.sort(pets, function(a, b) return (a.mps or 0) > (b.mps or 0) end)
    allPetsCache = pets
    return pets
end

local function getBestPet(mode)
    local pets = allPetsCache
    if #pets == 0 then return nil end
    if mode == "gen" then return pets[1]
    elseif mode == "value" then
        return pets[1]
    elseif mode == "priority" then
        for _, pname in ipairs(priorityList) do
            local s = pname:lower()
            for _, p in ipairs(pets) do
                if (p.name and p.name:lower() == s) or (p.index and p.index:lower() == s) then
                    return p
                end
            end
        end
        return pets[1]
    end
    return pets[1]
end

local InternalStealCache = {}
local STEAL_HOLD = 1.3

local function findStealPrompt(pet)
    if pet.plot and pet.slot then
        local plots = Workspace:FindFirstChild("Plots")
        local plot = plots and plots:FindFirstChild(pet.plot)
        local pods = plot and plot:FindFirstChild("AnimalPodiums")
        local pod = pods and pods:FindFirstChild(tostring(pet.slot))
        if pod then
            local base = pod:FindFirstChild("Base")
            local spawn = base and base:FindFirstChild("Spawn")
            local att = spawn and spawn:FindFirstChild("PromptAttachment")
            if att then for _, p in ipairs(att:GetChildren()) do if p:IsA("ProximityPrompt") then return p end end end
            for _, d in ipairs(pod:GetDescendants()) do if d:IsA("ProximityPrompt") then return d end end
        end
    end
    return nil
end
local function buildStealCallbacks(prompt)
    if InternalStealCache[prompt] then return end
    local data = {holdCallbacks = {}, triggerCallbacks = {}, holdEndCallbacks = {}, ready = true}
    local function grab(sig, into)
        local ok, conns = pcall(getconnections, sig)
        if ok and type(conns) == "table" then
            for _, c in ipairs(conns) do if type(c.Function) == "function" then table.insert(into, c.Function) end end
        end
    end
    grab(prompt.PromptButtonHoldBegan, data.holdCallbacks)
    grab(prompt.Triggered, data.triggerCallbacks)
    grab(prompt.PromptButtonHoldEnded, data.holdEndCallbacks)
    if #data.holdCallbacks > 0 or #data.triggerCallbacks > 0 or #data.holdEndCallbacks > 0 then
        InternalStealCache[prompt] = data
    end
end
local function executeStealAsync(prompt)
    local data = InternalStealCache[prompt]
    if not data or not data.ready then return false end
    data.ready = false
    task.spawn(function()
        for _, fn in ipairs(data.holdCallbacks) do task.spawn(fn) end
        pcall(function()
            local st = prompt:GetAttribute("State")
            if st ~= nil and st ~= "Steal" then
                if not _G._xenStealRemote then
                    _G._xenStealRemote = _G.Net.RemoteEvent(nil, "f40f7d9e-2f0d-4167-b250-899273f46874")
                end
                local r = _G._xenStealRemote
                if r then
                    local t = Workspace:GetServerTimeNow() + 124
                    r:FireServer(t, "68c86eb7-eb7e-4b4d-96ae-cf7cd847c5b0")
                    r:FireServer(t, "07b9cc25-2a1f-4a26-a0ec-f2fab578d8bd")
                end
            end
        end)
        task.wait(STEAL_HOLD)
        if prompt and prompt.Parent then for _, fn in ipairs(data.triggerCallbacks) do task.spawn(fn) end end
        for _, fn in ipairs(data.holdEndCallbacks) do task.spawn(fn) end
        task.wait(0.05)
        data.ready = true
    end)
    return true
end
local function autoStealLoop()
    while true do
        task.wait(0.1)
        if not Config.autoSteal then continue end
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then continue end
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            local st = hum:GetState()
            if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then continue end
        end
        local target = getBestPet(Config.targetMode)
        if not target then continue end
        local prompt = findStealPrompt(target)
        if prompt and prompt.Parent then
            local pp = prompt.Parent
            local ppPos = (pp:IsA("BasePart") and pp.Position) or (pp.Parent and pp.Parent:IsA("BasePart") and pp.Parent.Position)
            if ppPos and (hrp.Position - ppPos).Magnitude <= 65 then
                buildStealCallbacks(prompt)
                executeStealAsync(prompt)
            end
        end
    end
end

local function invis_createServerGhost(pos)
    if not invis_ghostEnabled or invis_errorOrbActive then return end
    local now = tick()
    if now - invis_lastLagbackTime < 0.05 then return end
    invis_lastLagbackTime = now
    if now - invis_lagbackWindowStart > 1 then
        invis_lagbackCallCount = 0
        invis_lagbackWindowStart = now
    end
    invis_lagbackCallCount = invis_lagbackCallCount + 1
    if invis_lagbackCallCount >= 7 then invis_createErrorOrb(); return end
    for _, g in pairs(invis_serverGhosts) do if g and g.Parent then g:Destroy() end end
    invis_serverGhosts = {}
    local ghost = Instance.new("Part")
    ghost.Name = "LagbackGhost"
    ghost.Shape = Enum.PartType.Ball
    ghost.Size = Vector3.new(3, 3, 3)
    ghost.Color = Color3.fromRGB(255, 0, 0)
    ghost.Material = Enum.Material.Glass
    ghost.Transparency = 0.3
    ghost.CanCollide = false
    ghost.Anchored = true
    ghost.CastShadow = false
    ghost.Position = pos + Vector3.new(0, 5, 0)
    ghost.Parent = Workspace.CurrentCamera
    table.insert(invis_serverGhosts, ghost)
end

local function invis_doClone()
    local c = player.Character
    if c and c:FindFirstChild("Humanoid") and c.Humanoid.Health > 0 then
        invis_hipHeight = c.Humanoid.HipHeight
        invis_oldRootPart = c:FindFirstChild("HumanoidRootPart")
        if not invis_oldRootPart or not invis_oldRootPart.Parent then return false end
        for _, v in pairs(invis_oldRootPart:GetChildren()) do
            if v:IsA("Attachment") and (v.Name:find("Beam") or v.Name:find("Attach")) then v:Destroy() end
        end
        for _, v in pairs(invis_oldRootPart:GetChildren()) do if v:IsA("Beam") then v:Destroy() end end
        local tmp = Instance.new("Model")
        tmp.Parent = game
        c.Parent = tmp
        invis_clonePart = invis_oldRootPart:Clone()
        invis_clonePart.Parent = c
        invis_oldRootPart.Parent = Workspace.CurrentCamera
        invis_clonePart.CFrame = invis_oldRootPart.CFrame
        c.PrimaryPart = invis_clonePart
        c.Parent = Workspace
        for _, v in pairs(c:GetDescendants()) do
            if v:IsA("Weld") or v:IsA("Motor6D") then
                if v.Part0 == invis_oldRootPart then v.Part0 = invis_clonePart end
                if v.Part1 == invis_oldRootPart then v.Part1 = invis_clonePart end
            end
        end
        tmp:Destroy()
        return true
    end
    return false
end
local function invis_revertClone()
    local c = player.Character
    if not invis_oldRootPart or not invis_oldRootPart:IsDescendantOf(Workspace) or not c or c.Humanoid.Health <= 0 then return end
    local tmp = Instance.new("Model")
    tmp.Parent = game
    c.Parent = tmp
    invis_oldRootPart.Parent = c
    c.PrimaryPart = invis_oldRootPart
    c.Parent = Workspace
    invis_oldRootPart.CanCollide = true
    for _, v in pairs(c:GetDescendants()) do
        if v:IsA("Weld") or v:IsA("Motor6D") then
            if v.Part0 == invis_clonePart then v.Part0 = invis_oldRootPart end
            if v.Part1 == invis_clonePart then v.Part1 = invis_oldRootPart end
        end
    end
    if invis_clonePart then
        local p = invis_clonePart.CFrame
        invis_clonePart:Destroy()
        invis_clonePart = nil
        invis_oldRootPart.CFrame = p
    end
    invis_oldRootPart = nil
    if c and c.Humanoid then c.Humanoid.HipHeight = invis_hipHeight end
    invis_clearAllGhosts()
end

local invis_turnOff, invis_turnOn
invis_turnOff = function()
    invis_clearAllGhosts()
    if not invis_animPlaying then return end
    local c = player.Character
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    invis_animPlaying = false
    for _, t in ipairs(invis_tracks) do pcall(function() t:Stop(0) end) end
    invis_tracks = {}
    if invis_connection then invis_connection:Disconnect(); invis_connection = nil end
    for _, cn in ipairs(invis_folderConnections) do if cn then cn:Disconnect() end end
    invis_folderConnections = {}
    if invis_antiDieConn then invis_antiDieConn:Disconnect(); invis_antiDieConn = nil end
    if invis_noCollisionConn then invis_noCollisionConn:Disconnect(); invis_noCollisionConn = nil end
    invis_revertClone()
    invis_clearAllGhosts()
    if hum then
        pcall(function()
            local anim = hum:FindFirstChildOfClass("Animator")
            if anim then
                for _, t in ipairs(anim:GetPlayingAnimationTracks()) do
                    if t.Priority == Enum.AnimationPriority.Action4 or t.Priority == Enum.AnimationPriority.Action3 then
                        t:Stop(0)
                    end
                end
            end
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            task.defer(function() if hum and hum.Parent then hum:ChangeState(Enum.HumanoidStateType.Running) end end)
        end)
    end
end
invis_turnOn = function()
    if invis_animPlaying then return end
    local c = player.Character
    if not c then return end
    local hum = c:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    invis_animPlaying = true
    invis_tracks = {}
    invis_removeFolders()
    if invis_doClone() then
        task.wait(0.05)
        invis_animationTrickery()
        local lastSetPos = nil
        local skipFrames = 5
        invis_connection = RunService.PreSimulation:Connect(function()
            if c and c:FindFirstChild("Humanoid") and c.Humanoid.Health > 0 and invis_oldRootPart then
                local root = c.PrimaryPart or c:FindFirstChild("HumanoidRootPart")
                if root then
                    if skipFrames > 0 then skipFrames = skipFrames - 1; lastSetPos = nil
                    elseif lastSetPos and invis_ghostEnabled then
                        local curPos = invis_oldRootPart.Position
                        if (curPos - lastSetPos).Magnitude > 6 and not invis_recoveryInProgress and player:GetAttribute("Stealing") then
                            lastSetPos = nil
                            invis_createServerGhost(curPos)
                            invis_recoveryInProgress = true
                            task.spawn(function()
                                pcall(invis_turnOff)
                                task.wait(0.6)
                                if player:GetAttribute("Stealing") then pcall(invis_turnOn) end
                                invis_recoveryInProgress = false
                            end)
                        end
                    end
                    if invis_clonePart then invis_clonePart.CanCollide = true end
                    if invis_oldRootPart and invis_oldRootPart.Parent then
                        for _, v in pairs(invis_oldRootPart:GetChildren()) do if v:IsA("Attachment") or v:IsA("Beam") then v:Destroy() end end
                        local sink = (Config.invisDepth or 7) * 0.5
                        invis_oldRootPart.CFrame = (root.CFrame - Vector3.new(0, sink, 0)) * CFrame.Angles(math.rad(Config.invisRotation or 225), 0, 0)
                        invis_oldRootPart.AssemblyLinearVelocity = root.AssemblyLinearVelocity
                        invis_oldRootPart.CanCollide = false
                        lastSetPos = invis_oldRootPart.Position
                    end
                end
            end
        end)
        invis_antiDieConn = hum:GetPropertyChangedSignal("Health"):Connect(function()
            if hum.Health <= 0 then hum.Health = hum.MaxHealth; hum:ChangeState(Enum.HumanoidStateType.Running) end
        end)
        invis_noCollisionConn = RunService.Stepped:Connect(function()
            if not invis_animPlaying then return end
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= player and p.Character then
                    for _, part in pairs(p.Character:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = false end end
                end
            end
        end)
    else
        invis_animPlaying = false
    end
end

            if hum.Health <= 0 then hum.Health = maxHp end
        end)
        char.AncestryChanged:Connect(function(_, parent)
            if not parent then conn1:Disconnect(); conn2:Disconnect() end
        end)
    end
    shieldCharacter(player.Character)
    player.CharacterAdded:Connect(shieldCharacter)
end

local BASES_LOW  = {[1] = Vector3.new(-476.52, -2, 220.94), [2] = Vector3.new(-476.52, -2, 113.77), [3] = Vector3.new(-476.52, -2, 6.18), [4] = Vector3.new(-476.52, -2, -101.07), [5] = Vector3.new(-342.66, -2, 221.45), [6] = Vector3.new(-342.66, -2, 113.41), [7] = Vector3.new(-342.66, -2, 6.25), [8] = Vector3.new(-342.66, -2, -99.73)}
local BASES_HIGH = {[1] = Vector3.new(-479.51, 18, 220.94), [2] = Vector3.new(-479.51, 18, 113.77), [3] = Vector3.new(-479.51, 18, 6.18), [4] = Vector3.new(-479.51, 18, -101.07), [5] = Vector3.new(-339.48, 18, 221.45), [6] = Vector3.new(-339.48, 18, 113.41), [7] = Vector3.new(-339.48, 18, 6.25), [8] = Vector3.new(-339.48, 18, -99.73)}
local FRONT_Y_HIGH = 17; local FRONT_Y_LOW = -2

local function getClosestBaseIdx(pos)
    local closest, dist = 1, math.huge
    for i = 1, 8 do
        local b = BASES_LOW[i]
        local d = (pos.X - b.X)^2 + (pos.Z - b.Z)^2
        if d < dist then dist = d; closest = i end
    end
    return closest
end
local function buildFrontCandidate(idx, isUpper, playerZ)
    local base   = isUpper and BASES_HIGH[idx] or BASES_LOW[idx]
    local frontY = isUpper and FRONT_Y_HIGH or FRONT_Y_LOW
    local frontZ = math.clamp(playerZ - base.Z, -FRONT_Z_CLAMP, FRONT_Z_CLAMP) + base.Z
    local faceDir = (idx <= 4) and Vector3.new(-1, 0, 0) or Vector3.new(1, 0, 0)
    return Vector3.new(base.X, frontY, frontZ), faceDir
end
local function plotSides(coordTable, idx)
    local base = BASES_LOW[idx]
    local isWest = idx <= 4
    local out = {}
    for _, coords in pairs(coordTable) do
        for _, data in ipairs(coords) do
            if ((data.coord.X < COLUMN_SPLIT_X) == isWest) and math.abs(data.coord.Z - base.Z) < SIDE_NEAR_Z then
                out[#out + 1] = {coord = data.coord, facing = data.facing}
            end
        end
    end
    return out
end

local function chooseStableApproachTarget(hrpPos, petPos, sideApproach, idx)
    local baseTable = (petPos.Y > 23.15) and UPPER or LOWER
    local baseIdx = idx or getClosestBaseIdx(petPos)
    local frontCoord, frontFace = buildFrontCandidate(baseIdx, baseTable == UPPER, hrpPos.Z)
    local sideCandidates = plotSides(baseTable, baseIdx)

    local function scorePosition(pos, label)
        local delta = hrpPos - pos
        local dist = delta.Magnitude
        local zBias = label == "side" and (math.abs(delta.Z) * 0.9) or 0
        local xBias = math.abs(delta.X) > 14 and 4 or 0
        local zClamp = math.abs(delta.Z) > 14 and 4 or 0
        return dist + zBias + xBias + zClamp
    end

    if sideApproach == "front" then
        return frontCoord, frontFace
    end

    if sideApproach == "side" then
        local bestDest, bestFace, bestScore = frontCoord, frontFace, math.huge
        for _, d in ipairs(sideCandidates) do
            local pos = d.coord
            local face = (d.facing == "NORTH") and Vector3.new(0, 0, -1) or Vector3.new(0, 0, 1)
            local score = scorePosition(pos, "side")
            if score < bestScore then
                bestScore = score
                bestDest = pos
                bestFace = face
            end
        end
        return bestDest, bestFace
    end

    local bestDest, bestFace, bestScore = frontCoord, frontFace, scorePosition(frontCoord, "front")
    for _, d in ipairs(sideCandidates) do
        local pos = d.coord
        local face = (d.facing == "NORTH") and Vector3.new(0, 0, -1) or Vector3.new(0, 0, 1)
        local score = scorePosition(pos, "side")
        if score < bestScore then
            bestScore = score
            bestDest = pos
            bestFace = face
        end
    end
    return bestDest, bestFace
end

local function spawnFollowPlatform(hrp, hum, targetY)

    local platY = targetY ~= nil and targetY or (hrp.Position.Y - hipH - 0.4)

    local plat = Instance.new("Part")
    plat.Name = "FollowPlat"
    plat.Size = Vector3.new(4, 0.55, 4)
    plat.Anchored = true
    plat.CanCollide = true
    plat.Material = Enum.Material.SmoothPlastic
    plat.Color = Color3.fromRGB(30, 120, 220)
    plat.Transparency = 1
    plat.CastShadow = false
    plat.Position = Vector3.new(hrp.Position.X, platY, hrp.Position.Z)
    plat.Parent = Workspace

    local active = true
    local followConn = RunService.Heartbeat:Connect(function()
        if not active or not hrp or not hrp.Parent or not plat or not plat.Parent then return end
        plat.Position = Vector3.new(hrp.Position.X, platY, hrp.Position.Z)
    end)

    local owConn = RunService.Stepped:Connect(function()
        if not active or not hrp or not hrp.Parent then return end
        plat.CanCollide = hrp.AssemblyLinearVelocity.Y <= 0.5 and hrp.Position.Y > platY
    end)

    local stealConn
    local function destroy()
        active = false
        followConn:Disconnect()
        owConn:Disconnect()
        if stealConn then stealConn:Disconnect() end
        if plat and plat.Parent then plat:Destroy() end
    end

    stealConn = player:GetAttributeChangedSignal("Stealing"):Connect(function()
        if player:GetAttribute("Stealing") then destroy() end
    end)

    return plat, destroy
end

local CHARACTER_RADIUS      = 4
local SAFE_RADIUS           = 3.1
local HARD_MIN_CLEARANCE    = 1.35
local LOOKAHEAD_DISTANCE    = 80
local MAX_PATH_ITERATIONS   = 30
local SIDE_DETOUR_DISTANCE  = 25
local MAX_VERTICAL_EFFORT   = 50
local SPLINE_SUBDIVISIONS   = 50
local WAYPOINT_ARRIVE_DIST  = 4
local VEL_BLEND             = 1.0
local REPLAN_STUCK_TIME     = 1.2

local SUPPORTS_SPHERECAST = false
pcall(function()
    local p = RaycastParams.new()
    Workspace:Spherecast(Vector3.new(0, 1e6, 0), 1, Vector3.new(0, 1, 0), p)
    SUPPORTS_SPHERECAST = true
end)

local function buildParams(ignoreList)
    local filter = {}
    if player.Character then table.insert(filter, player.Character) end
    for _, nm in ipairs({"NavTmpPart", "FollowPlat", "AnchorPlat"}) do
        local f = Workspace:FindFirstChild(nm)
        if f then table.insert(filter, f) end
    end
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= player and pl.Character then table.insert(filter, pl.Character) end
    end
    if ignoreList then
        for _, v in ipairs(ignoreList) do
            if typeof(v) == "Instance" then table.insert(filter, v) end
        end
    end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = filter
    params.IgnoreWater = true
    params.RespectCanCollide = true
    return params
end

local function basisOf(dir)
    local d = dir.Unit
    local up = (math.abs(d.Y) > 0.95) and Vector3.new(1, 0, 0) or Vector3.yAxis
    local right = d:Cross(up)
    if right.Magnitude < 1e-4 then right = Vector3.new(1, 0, 0) else right = right.Unit end
    local realUp = right:Cross(d).Unit
    return right, realUp
end

local function castVolume(origin, dir, dist, radius, params)
    if dist <= 0 then return nil end
    local d = dir
    if d.Magnitude < 1e-5 then return nil end
    d = d.Unit
    radius = radius or SAFE_RADIUS
    if SUPPORTS_SPHERECAST then
        local ok, res = pcall(function()
            return Workspace:Spherecast(origin, radius, d * dist, params)
        end)
        if ok then return res end
    end
    local right, up = basisOf(d)
    local diag = 0.7071 * radius
    local offs = {
        Vector3.zero,
        right * radius, -right * radius, up * radius, -up * radius,
        (right + up) * diag, (right - up) * diag, (-right + up) * diag, (-right - up) * diag,
    }
    local best
    for _, o in ipairs(offs) do
        local r = Workspace:Raycast(origin + o, d * dist, params)
        if r and (not best or r.Distance < best.Distance) then best = r end
    end
    return best
end

local function isSegmentClear(a, b, params, radius)
    local delta = b - a
    local dist = delta.Magnitude
    if dist < 1e-3 then return true end
    return castVolume(a, delta, dist, radius or SAFE_RADIUS, params) == nil
end

local function clearanceAt(pt, params)
    local dirs = {
        Vector3.new(1,0,0), Vector3.new(-1,0,0), Vector3.new(0,0,1), Vector3.new(0,0,-1),
        Vector3.new(0,1,0), Vector3.new(0,-1,0),
    }
    local minD = 30
    for _, d in ipairs(dirs) do
        local r = Workspace:Raycast(pt, d * 30, params)
        if r and r.Distance < minD then minD = r.Distance end
    end
    return minD
end

local function nudgeToOpenSpace(pt, params)
    local probe = {
        Vector3.new(1,0,0), Vector3.new(-1,0,0), Vector3.new(0,0,1), Vector3.new(0,0,-1),
        Vector3.new(0,1,0), Vector3.new(0,-1,0),
    }
    local push = Vector3.zero
    for _, d in ipairs(probe) do
        local r = Workspace:Raycast(pt, d * (SAFE_RADIUS + 2), params)
        if r then
            local pen = (SAFE_RADIUS + 2) - r.Distance
            push = push - d * pen
        end
    end
    if push.Magnitude > 0.05 then
        return pt + push
    end
    return pt
end

local function planPath(startPos, destination, ignoreList)
    local params = buildParams(ignoreList)
    local waypoints = { startPos }
    local current = startPos
    local iter = 0
    local visitedPenalty = {}

    while iter < MAX_PATH_ITERATIONS do
        local toDest = destination - current
        local dist = toDest.Magnitude
        if dist < WAYPOINT_ARRIVE_DIST + 1 then
            table.insert(waypoints, destination)
            break
        end

        local dir = toDest.Unit
        local lookOrigin = current + Vector3.new(0, CHARACTER_RADIUS * 0.25, 0)
        local scanLen = math.min(dist, LOOKAHEAD_DISTANCE)
        local hit = castVolume(lookOrigin, dir, scanLen, SAFE_RADIUS, params)

        if not hit then
            if scanLen >= dist then
                table.insert(waypoints, destination)
                break
            else
                local step = current + dir * (scanLen - 2)
                table.insert(waypoints, step)
                current = step
                iter = iter + 1
            end
        else
            local obstaclePos = hit.Position
            local inst = hit.Instance
            local obstacleTop = obstaclePos.Y + 4
            pcall(function()
                if inst and inst:IsA("BasePart") then
                    obstacleTop = inst.CFrame:PointToWorldSpace(Vector3.new(0, inst.Size.Y / 2, 0)).Y
                end
            end)
            local clearanceY   = obstacleTop + CHARACTER_RADIUS + 2
            local ascentNeeded = clearanceY - current.Y
            local canGoOver    = ascentNeeded <= MAX_VERTICAL_EFFORT

            local rightDir = (CFrame.fromAxisAngle(Vector3.yAxis, math.rad(90)) * dir)
            rightDir = Vector3.new(rightDir.X, 0, rightDir.Z)
            if rightDir.Magnitude < 1e-3 then rightDir = Vector3.new(1, 0, 0) end
            rightDir = rightDir.Unit
            local leftDir = -rightDir

            local candidates = {}
            local function consider(pt, extraCost)
                pt = nudgeToOpenSpace(pt, params)
                if not isSegmentClear(current, pt, params) then return end
                local key = string.format("%d_%d_%d", math.floor(pt.X/4), math.floor(pt.Y/4), math.floor(pt.Z/4))
                local pen = visitedPenalty[key] and 400 or 0
                local openness = clearanceAt(pt, params)
                local cost = (pt - destination).Magnitude + (extraCost or 0) + pen + math.max(0, (12 - openness)) * 3
                table.insert(candidates, { pt = pt, cost = cost, key = key })
            end

            if canGoOver then
                consider(Vector3.new(obstaclePos.X, clearanceY, obstaclePos.Z) + dir * 6, ascentNeeded * 0.5)
            end
            for _, mult in ipairs({ 0.6, 1.0, 1.6 }) do
                for _, sideDir in ipairs({ rightDir, leftDir }) do
                    consider(current + sideDir * (SIDE_DETOUR_DISTANCE * mult) + dir * 8, SIDE_DETOUR_DISTANCE * mult * 0.25)
                end
            end
            if canGoOver then
                for _, sideDir in ipairs({ rightDir, leftDir }) do
                    consider(current + sideDir * SIDE_DETOUR_DISTANCE + Vector3.new(0, ascentNeeded, 0) + dir * 8,
                             ascentNeeded * 0.5 + SIDE_DETOUR_DISTANCE * 0.5)
                end
            end
            consider(current + Vector3.new(0, -14, 0) + dir * 8, 30)
            consider(current + Vector3.new(0, 26, 0), 45)

            if #candidates == 0 then
                local up = Vector3.new(0, 1, 0)
                local upHit = castVolume(current, up, 30, SAFE_RADIUS, params)
                local rise = upHit and math.max(0, upHit.Distance - SAFE_RADIUS - 1) or 30
                if rise < 2 then
                    local backHit = castVolume(current, -dir, 20, SAFE_RADIUS, params)
                    local back = backHit and math.max(0, backHit.Distance - SAFE_RADIUS - 1) or 20
                    if back < 2 then break end
                    current = current - dir * back
                else
                    current = current + up * rise
                end
                table.insert(waypoints, current)
            else
                table.sort(candidates, function(a, b) return a.cost < b.cost end)
                local best = candidates[1]
                visitedPenalty[best.key] = true
                table.insert(waypoints, best.pt)
                current = best.pt
            end

            iter = iter + 1
        end
    end

    if (waypoints[#waypoints] - destination).Magnitude > 0.01 then
        table.insert(waypoints, destination)
    end

    local pulled = { waypoints[1] }
    local i = 1
    while i < #waypoints do
        local anchor = pulled[#pulled]
        local furthest = i + 1
        for j = #waypoints, i + 1, -1 do
            if isSegmentClear(anchor, waypoints[j], params) then furthest = j break end
        end
        table.insert(pulled, waypoints[furthest])
        i = furthest
    end

    return pulled, params
end

local function catmullRom(p0, p1, p2, p3, t)
    local t2, t3 = t * t, t * t * t
    return 0.5 * (
        (2 * p1) +
        (-p0 + p2) * t +
        (2 * p0 - 5 * p1 + 4 * p2 - p3) * t2 +
        (-p0 + 3 * p1 - 3 * p2 + p3) * t3
    )
end

local function smoothPath(pts, params)
    if #pts < 3 then return pts end
    local out = { pts[1] }
    for i = 1, #pts - 1 do
        local p0 = pts[math.max(i - 1, 1)]
        local p1 = pts[i]
        local p2 = pts[i + 1]
        local p3 = pts[math.min(i + 2, #pts)]
        local segment = {}
        local valid = true
        local prev = p1
        for s = 1, SPLINE_SUBDIVISIONS do
            local t = s / SPLINE_SUBDIVISIONS
            local pt = catmullRom(p0, p1, p2, p3, t)
            if not isSegmentClear(prev, pt, params) then valid = false break end
            table.insert(segment, pt)
            prev = pt
        end
        if valid then
            for _, pt in ipairs(segment) do table.insert(out, pt) end
        else
            table.insert(out, p2)
        end
    end
    if (out[#out] - pts[#pts]).Magnitude > 0.01 then table.insert(out, pts[#pts]) end
    return out
end

local FEELERS = {}
do
    for _, yaw in ipairs({ 0, 22, -22, 45, -45, 70, -70 }) do
        table.insert(FEELERS, { yaw = math.rad(yaw), pitch = 0 })
    end
    table.insert(FEELERS, { yaw = 0, pitch = math.rad(30) })
    table.insert(FEELERS, { yaw = 0, pitch = math.rad(-30) })
end

local function avoidanceVector(origin, travelDir, speed, params)
    local probeLen = math.clamp(speed * 0.10, 8, 32)
    local right, up = basisOf(travelDir)
    local steer = Vector3.zero
    local nearest = math.huge
    for _, f in ipairs(FEELERS) do
        local d = (CFrame.fromAxisAngle(Vector3.yAxis, f.yaw) * travelDir)
        if f.pitch ~= 0 then d = (CFrame.fromAxisAngle(right, f.pitch) * d) end
        d = d.Unit
        local hit = castVolume(origin, d, probeLen, SAFE_RADIUS * 0.9, params)
        if hit then
            local weight = 1 - (hit.Distance / probeLen)
            steer = steer + hit.Normal * (weight * weight) * 1.2
            local tangent = hit.Normal:Cross(Vector3.yAxis)
            if tangent.Magnitude > 1e-3 then
                tangent = tangent.Unit
                if tangent:Dot(right) < 0 then tangent = -tangent end
                steer = steer + tangent * weight * 1.1
            end
            if hit.Distance < nearest then nearest = hit.Distance end
        end
    end
    return steer, nearest
end

local function predictiveObstacleFly(hrp, destination, speed, onTick, ignoreList)
    if not hrp or not hrp.Parent then
        return
    end

    local raw, params = planPath(hrp.Position, destination, ignoreList)
    local waypoints = smoothPath(raw, params)
    local wpIndex = 2
    local stuckTimer = 0
    local lastPos = hrp.Position
    local currentVel = hrp.AssemblyLinearVelocity
    local flightStart = os.clock()
    local replanCooldown = 0

    while hrp and hrp.Parent do
        if onTick then pcall(onTick) end
        local dt = RunService.Heartbeat:Wait()
        if not (hrp and hrp.Parent) then break end
        if os.clock() - flightStart > 45 then
            break
        end
        replanCooldown = math.max(0, replanCooldown - dt)

        local myPos = hrp.Position
        if (myPos - destination).Magnitude < 3.5 then
            break
        end

        if wpIndex > #waypoints then break end
        local targetWP = waypoints[wpIndex]

        while wpIndex < #waypoints do
            local d = (waypoints[wpIndex] - myPos).Magnitude
            if d < WAYPOINT_ARRIVE_DIST then
                wpIndex = wpIndex + 1
            else
                break
            end
        end
        targetWP = waypoints[math.min(wpIndex, #waypoints)]

        local diff = targetWP - myPos
        local dist = diff.Magnitude
        if dist < 0.05 then
            wpIndex = wpIndex + 1
            continue
        end
        local seekDir = diff.Unit

        local blended = seekDir

        local speedScale = 1
        local arriveScale = math.clamp(dist / 20, 0.5, 1)
        if wpIndex >= #waypoints then
            speedScale = arriveScale
        end

        local desiredVel = blended * (speed * speedScale)
        currentVel = desiredVel
        hrp.AssemblyLinearVelocity = desiredVel
        hrp.AssemblyAngularVelocity = Vector3.zero

        local posDelta = (myPos - lastPos).Magnitude
        if posDelta < 0.5 then
            stuckTimer = stuckTimer + dt
        else
            stuckTimer = 0
            lastPos = myPos
        end
        if stuckTimer > REPLAN_STUCK_TIME then
            raw, params = planPath(myPos, destination, ignoreList)
            waypoints = smoothPath(raw, params)
            wpIndex = 2
            stuckTimer = 0
            lastPos = myPos
            local upHit = castVolume(myPos, Vector3.yAxis, 24, SAFE_RADIUS, params)
            if not upHit then currentVel = Vector3.new(0, speed * 0.6, 0) end
        end
    end
    vZero(hrp)
end

local function directFlyTo(hrp, destination, speed, onTick, ignoreList)
    if not hrp or not hrp.Parent then return end
    local params = buildParams(ignoreList)
    local currentVel = hrp.AssemblyLinearVelocity
    local stuckTimer = 0
    local lastPos = hrp.Position
    local t0 = os.clock()

    while hrp and hrp.Parent do
        if onTick then pcall(onTick) end
        local dt = RunService.Heartbeat:Wait()
        if not (hrp and hrp.Parent) then break end
        if os.clock() - t0 > 25 then break end

        local myPos = hrp.Position
        local diff = destination - myPos
        local dist = diff.Magnitude
        if dist < 2 then
            break
        end
        local dir = diff.Unit

        local arriveScale = math.clamp(dist / 12, 0.3, 1)
        local desiredVel = dir * (speed * arriveScale)
        currentVel = currentVel:Lerp(desiredVel, 0.25)

        local frameMove = currentVel * math.max(dt, 1 / 60)
        local moveLen = frameMove.Magnitude
        if moveLen > 1e-4 then
            local hit = castVolume(myPos, frameMove, moveLen + HARD_MIN_CLEARANCE, SAFE_RADIUS * 0.8, params)
            if hit then
                local allowed = math.max(0, hit.Distance - HARD_MIN_CLEARANCE)
                if allowed <= 0.05 then
                    local n = hit.Normal
                    local slide = currentVel - n * currentVel:Dot(n)
                    currentVel = (slide.Magnitude > 1) and slide.Unit * math.min(slide.Magnitude, speed * 0.5) or Vector3.zero
                else
                    local maxSpeed = allowed / math.max(dt, 1 / 60)
                    if currentVel.Magnitude > maxSpeed then currentVel = currentVel.Unit * maxSpeed end
                end
            end
        end

        hrp.AssemblyLinearVelocity = currentVel
        hrp.AssemblyAngularVelocity = Vector3.zero

        if (myPos - lastPos).Magnitude < 0.8 then
            stuckTimer = stuckTimer + dt
        else
            stuckTimer = 0
            lastPos = myPos
        end
        if stuckTimer > 2 then
            local upHit = castVolume(myPos, Vector3.yAxis, 48, SAFE_RADIUS, params)
            local rise = upHit and math.max(0, upHit.Distance - SAFE_RADIUS - 1) or 45
            if rise > 3 then
                hrp.AssemblyLinearVelocity = Vector3.new(0, speed, 0)
                local riseStart = myPos.Y
                while hrp and hrp.Parent do
                    if hrp.Position.Y - riseStart > math.min(45, rise) then break end
                    RunService.Heartbeat:Wait()
                end
            else
                predictiveObstacleFly(hrp, destination, speed, onTick, ignoreList)
                return
            end
            stuckTimer = 0
            lastPos = hrp.Position
        end
    end
    vZero(hrp)
end

local function disableAllAnimations(char)
    if not char then return end
    local animate = char:FindFirstChild("Animate")
    if animate then animate.Disabled = true end
    for _, animator in ipairs(char:GetDescendants()) do
        if animator:IsA("Animator") then
            pcall(function()
                for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                    track:Stop(0)
                end
            end)
        end
    end
end

local function enableAllAnimations(char)
    if not char then return end
    local animate = char:FindFirstChild("Animate")
    if animate then animate.Disabled = false end
end

local isTeleporting = false
local lastTargetData = nil

local function executeTP()
    if isTeleporting then return end
    isTeleporting = true
    autoBackActive = false
    if autoBackCoroutine then autoBackCoroutine = nil end

    if Config.tpDelay and Config.tpDelay > 0 then
        task.wait(Config.tpDelay)
    end

    local char = player.Character or player.CharacterAdded:Wait()
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then isTeleporting = false; return end
    if hum.Health <= 0 then setStatus("Died – aborting"); isTeleporting = false; return end

    disableAllAnimations(char)
    pcall(disableHRPAntiCheat)

    local safetyPlat = Instance.new("Part")
    safetyPlat.Name = "NavTmpPart"; safetyPlat.Size = Vector3.new(8, 1, 8); safetyPlat.Position = Vector3.new(hrp.Position.X, hrp.Position.Y-5, hrp.Position.Z)
    safetyPlat.Anchored = true; safetyPlat.CanCollide = true; safetyPlat.Transparency = 1; safetyPlat.Material = Enum.Material.SmoothPlastic; safetyPlat.Parent = Workspace
    task.delay(14, function() if safetyPlat and safetyPlat.Parent then safetyPlat:Destroy() end end)

    local function cleanup()
        if safetyPlat and safetyPlat.Parent then safetyPlat:Destroy() end
        isTeleporting = false
        if char and char.Parent then enableAllAnimations(char) end
    end

    local ok, err = pcall(function()
        local target = getTargetPetData()
        if not target then setStatus("No target"); cleanup(); return end

        local petPos   = target.position; local petName  = target.name; local plotName = target.plot
        setStatus("→ " .. petName)
        currentTargetFloorY = petPos.Y

        local speed       = Config.tpMode == "grapple" and Config.grappleTweenSpeed or Config.carpetTweenSpeed
        local cloneDelay  = Config.tpMode == "grapple" and Config.grappleCloneDelay  or Config.carpetCloneDelay
        local isLowerFloor = petPos.Y <= 8.9
        local squareSz    = isLowerFloor and PLOT_SQUARE_SIZE_FIRST or PLOT_SQUARE_SIZE_SECOND
        lastTargetData = { petPos = petPos, plotName = plotName, squareSize = squareSz, uid = target.uid }

        local isFirstFloor = petPos.Y <= 8.9
        local isUnlocked = isFirstFloor and _G._isTargetPlotUnlocked(plotName)

        if isUnlocked then
            setStatus("Unlocked – direct entry")
            if Config.tpMode == "grapple" then carpetEngage() else equipCarpet() end
            vZero(hrp)

            local idx = getClosestBaseIdx(petPos)
            local frontPos, frontFace = buildFrontCandidate(idx, false, hrp.Position.Z)
            local frontDest = frontPos + Vector3.new(0, 0.15, 0)

            local plotModel = Workspace:FindFirstChild("Plots") and Workspace.Plots:FindFirstChild(plotName)
            local ignoreList = {Workspace:FindFirstChild("NavTmpPart")}
            if plotModel then table.insert(ignoreList, plotModel) end

            predictiveObstacleFly(hrp, frontDest, math.max(speed * 0.85, 220), equipCarpet, ignoreList)
            if (hrp.Position - frontDest).Magnitude > 1 then
                directFlyTo(hrp, frontDest, math.max(speed * 0.7, 200), equipCarpet, ignoreList)
            end
            vZero(hrp)

            local adjY = petPos.Y
            if TALL_PETS[petName] then adjY = petPos.Y - TALL_OFFSET end
            local targetY = getTargetHeight(adjY)
            local dest = Vector3.new(petPos.X, targetY, petPos.Z)
            directFlyTo(hrp, dest, speed, equipCarpet)

            if hrp and hrp.Parent then
                hrp.CFrame = CFrame.new(dest)
                hrp.AssemblyLinearVelocity = Vector3.zero
            end
            setStatus("Arrived – free to steal")
            startAutoBackBrainrot(lastTargetData)
            cleanup()
            return
        end

        if Config.tpMode == "grapple" then carpetEngage() else equipCarpet() end
        vZero(hrp)

        local adjY = petPos.Y
        if TALL_PETS[petName] then adjY = petPos.Y - TALL_OFFSET end

        local function getPlotEdge(plotName, hrpPos, petPos, sideApproach)
            local plots = Workspace:FindFirstChild("Plots")
            if not plots then return nil, nil end
            local plot = plots:FindFirstChild(plotName)
            if not plot then return nil, nil end
            local center, size
            pcall(function()
                local bbox = plot:GetBoundingBox()
                center = bbox.Position
                size = bbox.Size
            end)
            if not center or not size then
                local idx = getClosestBaseIdx(petPos)
                local coordTable = petPos.Y > UPPER_Y_THRESHOLD and UPPER or LOWER
                local isUpper = coordTable == UPPER
                local frontCoord, frontFace = buildFrontCandidate(idx, isUpper, hrpPos.Z)
                return frontCoord, frontFace
            end
            local dirToCenter = (center - Vector3.new(hrpPos.X, 0, hrpPos.Z)).Unit
            local halfX = size.X / 2
            local halfZ = size.Z / 2
            local edgePos
            if sideApproach == "front" or sideApproach == "auto" then
                local frontZ = center.Z + (hrpPos.Z < center.Z and halfZ or -halfZ)
                edgePos = Vector3.new(center.X, center.Y, frontZ)
            else
                local sideX = center.X + (hrpPos.X < center.X and halfX or -halfX)
                edgePos = Vector3.new(sideX, center.Y, center.Z)
            end
            edgePos = Vector3.new(edgePos.X, petPos.Y > 23.15 and 17 or -2, edgePos.Z)
            local faceDir = (center - edgePos).Unit
            return edgePos, faceDir
        end

        local approachDest, approachFace = getPlotEdge(plotName, hrp.Position, petPos, Config.sideApproach)
        if not approachDest then
            local idx = getClosestBaseIdx(petPos)
            local coordTable = petPos.Y > UPPER_Y_THRESHOLD and UPPER or LOWER
            local isUpper = coordTable == UPPER
            approachDest, approachFace = buildFrontCandidate(idx, isUpper, hrp.Position.Z)
        end
        local stableDest = approachDest + Vector3.new(0, 0.15, 0)

        local plotModel = Workspace:FindFirstChild("Plots") and Workspace.Plots:FindFirstChild(plotName)
        local approachIgnore = {Workspace:FindFirstChild("NavTmpPart")}
        if plotModel then table.insert(approachIgnore, plotModel) end

        setStatus("Stabilizing approach...")
        predictiveObstacleFly(hrp, stableDest, math.max(speed * 0.85, 220), equipCarpet, approachIgnore)

        local settleDist = (hrp.Position - stableDest).Magnitude
        if settleDist > 1 then
            directFlyTo(hrp, stableDest, math.max(speed * 0.7, 200), equipCarpet, approachIgnore)
        end

        vZero(hrp)

        local exactEdgePos = approachDest
        local faceVec = approachFace or Vector3.new(0, 0, -1)
        local finalPos = exactEdgePos + faceVec * 0.0
        local targetCF = CFrame.new(finalPos, finalPos + faceVec)
        hrp.CFrame = targetCF
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
        for i = 1, 8 do
            hrp.CFrame = targetCF
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
            task.wait(0.01)
        end

        local finalOffset = (approachFace or Vector3.new(0,0,-1)) * 0.2
        local finalPos2 = stableDest + finalOffset
        local finalPos2CF = CFrame.new(finalPos2, finalPos2 + (approachFace or Vector3.new(0,0,-1)))

        local hipH = hum.HipHeight or 2.3
        local stopPlat = Instance.new("Part")
        stopPlat.Name = "AnchorPlat"
        stopPlat.Size = Vector3.new(2.6, 0.3, 2.6)
        stopPlat.Anchored = true
        stopPlat.CanCollide = true
        stopPlat.Transparency = 1
        stopPlat.Material = Enum.Material.SmoothPlastic
        stopPlat.Position = Vector3.new(stableDest.X, stableDest.Y - hipH - 0.1, stableDest.Z)
        stopPlat.Parent = Workspace

        hrp.Anchored = true
        setStatus("Anchored – waiting clone...")

        if cloneDelay > 0 then
            task.wait(cloneDelay)
            hrp.Anchored = false
            hrp.CFrame = finalPos2CF
            vZero(hrp)
            task.wait(0.05)
            hrp.Anchored = true
        end

        local cloneSuccess = false
        local retryCount = 0
        local startPos = hrp.Position

        while retryCount < 2 and not cloneSuccess do
            if retryCount > 0 then
                hrp.Anchored = false
                hrp.CFrame = finalPos2CF
                vZero(hrp)
                task.wait(0.1)
                hrp.Anchored = true
                task.wait(0.05)
            end
            doClone()
            local waitStart = os.clock()
            while os.clock() - waitStart < 4 and hrp.Parent do
                if (hrp.Position - startPos).Magnitude > 2.5 then cloneSuccess = true; break end
                task.wait()
            end
            if not cloneSuccess then retryCount = retryCount + 1; startPos = hrp.Position end
        end

        if hrp and hrp.Parent then hrp.Anchored = false end
        if stopPlat and stopPlat.Parent then stopPlat:Destroy() end

        if not cloneSuccess then setStatus("Clone failed – aborting"); cleanup(); return end
        if not isInsidePlotSquare(hrp.Position, plotName, squareSz) then setStatus("Clone missed base – aborting"); cleanup(); return end

        task.wait(0.15)
        char = player.Character; if not char then cleanup(); return end
        hrp = char:FindFirstChild("HumanoidRootPart"); hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum then cleanup(); return end

        setStatus("Teleporting to pet...")
        local targetY = getTargetHeight(adjY)
        lastTargetData.targetY = targetY
        local dest = Vector3.new(petPos.X, targetY, petPos.Z)

        local followPlat = nil
        local function destroyPlat()
            if followPlat and followPlat.Parent then followPlat:Destroy() end
        end
        if petPos.Y > 23.15 then
            followPlat = Instance.new("Part")
            followPlat.Name = "FollowPlat"
            followPlat.Size = Vector3.new(5, 0.5, 5)
            followPlat.Anchored = true
            followPlat.CanCollide = true
            followPlat.Material = Enum.Material.SmoothPlastic
            followPlat.Color = Color3.fromRGB(30, 120, 220)
            followPlat.Transparency = 0.5
            followPlat.CastShadow = false
            followPlat.Position = Vector3.new(hrp.Position.X, dest.Y - 0.5, hrp.Position.Z)
            followPlat.Parent = Workspace
            task.delay(5, destroyPlat)
        end

        if hrp and hrp.Parent then
            hrp.CFrame = CFrame.new(dest)
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end

        if followPlat then task.delay(30, destroyPlat) end
        setStatus("Arrived – free to steal")
        startAutoBackBrainrot(lastTargetData)
    end)

    cleanup()
    if not ok then setStatus("Err: " .. tostring(err):sub(1,32)) end
end
